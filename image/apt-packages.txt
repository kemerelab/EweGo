# Debian / Raspberry Pi OS packages installed inside the image at build time
# (image/inject-ewego.sh runs apt-get in an emulated chroot). One per line.
# python3-picamera2 is deliberately absent: the cameras are USB (UVC) and
# recorded by ewego-cam; the Picamera2/CSI scripts stay in Firmware/ for
# reference only. smbus2 = fuel gauge, pyserial = GPS logger.
python3-smbus2
python3-serial
i2c-tools
psmisc
# ffmpeg for ad-hoc streaming/transcoding tests with USB cameras (not used
# by the recording scripts).
ffmpeg
# dfu-util flashes the hub_imu (STM32F042) USB IMU module through the hub
# via the STM32 system bootloader (hold BOOT0 high at plug-in -> 0483:df11).
dfu-util
# avahi-daemon gives <hostname>.local. Raspberry Pi Imager's cloud-init
# user-data would otherwise apt-install it at first boot; the image disables
# that module, so it has to be here.
avahi-daemon
